```typescript
/**
 * Feedback Generation - Bible Section 8.5
 * 
 * Generates actionable feedback for implementor after verification failure.
 * CRITICAL: Must not reveal hidden verification criteria.
 */

import * as fs from "node:fs";
import * as path from "node:path";
import { loadManifest, getTask } from "./manifest.js";
import { loadProgress, updateProgress } from "./progress.js";
import { loadConfig, getResolvedPaths, requireOrchestraRoot } from "./config.js";
import { renderTemplate } from "./templates.js";
import { VerifyResult, VerifyCheckResult } from "./verification.js";
import { OrchestraError, ErrorCode } from "./errors.js";

export interface FeedbackOptions {
  task?: string;
  verificationResult?: VerifyResult;
  attempt?: number;
  orchestraRoot?: string;
}

export interface FeedbackIssue {
  severity: "critical" | "major" | "minor";
  category: string;
  problem: string;
  impact: string;
  guidance: string;
}

export interface FeedbackResult {
  success: boolean;
  taskId: number;
  attempt: number;
  maxAttempts: number;
  canRetry: boolean;
  feedbackPath: string;
  issues: FeedbackIssue[];
  nextStep: "retry" | "escalate";
}

/**
 * Transform verification failures into implementor-friendly feedback.
 * CRITICAL: Strip any information that reveals HOW we detected the issue.
 */
function transformToFeedback(checks: VerifyCheckResult[]): FeedbackIssue[] {
  return checks
    .filter((c) => !c.passed)
    .map((check) => ({
      severity: check.severity as "critical" | "major" | "minor",
      category: check.category || "general",
      // Use description, NOT the check details
      problem: check.description,
      impact: check.impact || "Verification cannot pass",
      // Guidance from check, or generic
      guidance: check.guidance || "Please review and fix this issue",
    }));
}

export async function runFeedback(
  options: FeedbackOptions
): Promise<FeedbackResult> {
  const orchestraRoot = options.orchestraRoot || requireOrchestraRoot();
  const config = loadConfig(orchestraRoot);
  const paths = getResolvedPaths(orchestraRoot, config);

  // Load manifest and progress
  const manifestResult = loadManifest(paths.manifest);
  if (!manifestResult.success || !manifestResult.data) {
    throw new OrchestraError(ErrorCode.CONFIG_ERROR, "Cannot load manifest");
  }

  const progressResult = loadProgress(paths.progress);
  if (!progressResult.success || !progressResult.data) {
    throw new OrchestraError(ErrorCode.CONFIG_ERROR, "Cannot load progress");
  }

  // Determine task ID
  const taskId = options.task
    ? parseInt(options.task, 10)
    : progressResult.data.currentTask;

  if (!taskId) {
    throw new OrchestraError(ErrorCode.VALIDATION_ERROR, "No task specified");
  }

  const task = getTask(manifestResult.data, taskId);
  if (!task) {
    throw new OrchestraError(
      ErrorCode.VALIDATION_ERROR,
      `Task ${taskId} not found`
    );
  }

  // Load or use provided verification result
  let verifyResult = options.verificationResult;
  if (!verifyResult) {
    const verifyPath = path.join(
      paths.artifacts,
      `task-${taskId}`,
      "verification.yaml"
    );
    if (!fs.existsSync(verifyPath)) {
      throw new OrchestraError(
        ErrorCode.VALIDATION_ERROR,
        "No verification results found. Run 'orchestra verify' first."
      );
    }
    // Load from file (implement readYaml call)
  }

  // Calculate attempt
  const taskProgress = progressResult.data.tasks?.find((t) => t.id === taskId);
  const currentAttempt = options.attempt || (taskProgress?.attempts || 0) + 1;
  const maxAttempts = config.verification?.maxAttempts || 3;
  const canRetry = currentAttempt < maxAttempts;

  // Transform to feedback (strips verification details)
  const issues = verifyResult
    ? transformToFeedback(verifyResult.checks)
    : [];

  // Render feedback template
  const feedbackContent = await renderTemplate("feedback", {
    taskId,
    task,
    attempt: currentAttempt,
    maxAttempts,
    issues,
    canRetry,
    timestamp: new Date().toISOString(),
  });

  // Write feedback file
  const feedbackPath = path.join(
    paths.feedback,
    `task-${taskId}-feedback.md`
  );
  fs.mkdirSync(path.dirname(feedbackPath), { recursive: true });
  fs.writeFileSync(feedbackPath, feedbackContent);

  // Update progress with incremented attempt
  // (implement updateProgress call)

  return {
    success: true,
    taskId,
    attempt: currentAttempt,
    maxAttempts,
    canRetry,
    feedbackPath,
    issues,
    nextStep: canRetry ? "retry" : "escalate",
  };
}
```

### CLI Command: src/commands/feedback.ts

```typescript
import { Command } from "commander";
import { runFeedback } from "../core/feedback.js";
import * as output from "../core/output.js";

export function createFeedbackCommand(): Command {
  const cmd = new Command("feedback")
    .description("Generate feedback for implementor after verification failure")
    .option("-t, --task <id>", "Task ID (defaults to current)")
    .option("-a, --attempt <n>", "Attempt number", parseInt)
    .option("--json", "Output as JSON")
    .action(async (options) => {
      try {
        const result = await runFeedback({
          task: options.task,
          attempt: options.attempt,
        });

        if (options.json) {
          console.log(JSON.stringify(result, null, 2));
        } else {
          output.success(
            `Feedback generated for task ${result.taskId} (attempt ${result.attempt} of ${result.maxAttempts})`
          );
          
          if (result.issues.length > 0) {
            console.log("\nIssues found:");
            result.issues.forEach((issue, i) => {
              console.log(
                `  ${i + 1}. [${issue.severity.toUpperCase()}] ${issue.problem}`
              );
            });
          }
          
          console.log(`\nFeedback written to: ${result.feedbackPath}`);
          
          if (!result.canRetry) {
            output.warn(
              "\nMaximum attempts reached. Consider escalating:\n" +
              `  orchestra escalate --task ${result.taskId} --reason "Persistent failure"`
            );
          } else {
            console.log("\nNext: Implementor should read feedback and retry");
          }
        }
      } catch (error) {
        output.error(error instanceof Error ? error.message : String(error));
        process.exit(1);
      }
    });

  return cmd;
}
```
