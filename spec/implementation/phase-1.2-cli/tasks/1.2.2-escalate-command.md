/**
 * Escalation Handling - Bible Section 8.5
 * 
 * Escalates persistent failures to human supervisor.
 * Halts workflow until human intervention.
 */

import * as fs from "node:fs";
import * as path from "node:path";
import { loadManifest, getTask, updateTaskStatus } from "./manifest.js";
import { loadProgress, updateProgress } from "./progress.js";
import { loadConfig, getResolvedPaths, requireOrchestraRoot } from "./config.js";
import { OrchestraError, ErrorCode } from "./errors.js";

export interface EscalateOptions {
  task?: string;
  reason: string;
  context?: string;
  orchestraRoot?: string;
}

export interface AttemptRecord {
  attempt: number;
  outcome: "PASS" | "FAIL";
  issue?: string;
  timestamp: string;
}

export interface EscalateResult {
  success: boolean;
  taskId: number;
  previousStatus: string;
  newStatus: "ESCALATED";
  reason: string;
  context?: string;
  escalatedAt: string;
  reportPath: string;
  attemptHistory: AttemptRecord[];
  humanOptions: string[];
}

/**
 * Compile attempt history from feedback files and progress
 */
function compileAttemptHistory(
  feedbackDir: string,
  taskId: number
): AttemptRecord[] {
  const history: AttemptRecord[] = [];
  
  // Find all feedback files for this task
  if (fs.existsSync(feedbackDir)) {
    const files = fs.readdirSync(feedbackDir)
      .filter((f) => f.startsWith(`task-${taskId}-`) && f.endsWith(".md"))
      .sort();
    
    files.forEach((file, index) => {
      const content = fs.readFileSync(path.join(feedbackDir, file), "utf-8");
      // Extract issue summary from content (first issue line)
      const issueMatch = content.match(/\*\*Problem\*\*: (.+)/);
      history.push({
        attempt: index + 1,
        outcome: "FAIL",
        issue: issueMatch?.[1] || "Unknown issue",
        timestamp: fs.statSync(path.join(feedbackDir, file)).mtime.toISOString(),
      });
    });
  }
  
  return history;
}

/**
 * Generate escalation report content
 */
function generateEscalationReport(
  task: any,
  reason: string,
  context: string | undefined,
  attemptHistory: AttemptRecord[],
  feedbackFiles: string[]
): string {
  const lines: string[] = [
    `# Escalation Report: Task ${task.id}`,
    "",
    "## Summary",
    `- **Task**: ${task.title}`,
    `- **Escalated**: ${new Date().toISOString()}`,
    `- **Reason**: ${reason}`,
    `- **Attempts**: ${attemptHistory.length}`,
    "",
  ];

  if (context) {
    lines.push("## Additional Context", "", context, "");
  }

  lines.push(
    "## Task Details",
    "",
    task.description || "(No description)",
    "",
    "## Attempt History",
    "",
    "| Attempt | Date | Outcome | Issue |",
    "|---------|------|---------|-------|"
  );

  attemptHistory.forEach((a) => {
    lines.push(
      `| ${a.attempt} | ${a.timestamp.split("T")[0]} | ${a.outcome} | ${a.issue || "-"} |`
    );
  });

  lines.push(
    "",
    "## Feedback Given",
    "",
    "See feedback files:",
    ...feedbackFiles.map((f) => `- ${f}`),
    "",
    "## Human Options",
    "",
    "- [ ] **Fix manually and complete**: Make changes, verify, then `orchestra complete --task " + task.id + " --force`",
    "- [ ] **Modify task specification**: Edit manifest.yaml, reset attempts, retry",
    "- [ ] **Skip this task**: `orchestra complete --task " + task.id + " --skip --reason \"...\"`",
    "- [ ] **Abort sprint**: Update progress.yaml status to `aborted`",
    "",
    "---",
    "_End of Escalation Report_"
  );

  return lines.join("\n");
}

export async function runEscalate(
  options: EscalateOptions
): Promise<EscalateResult> {
  if (!options.reason) {
    throw new OrchestraError(
      ErrorCode.VALIDATION_ERROR,
      "--reason is required"
    );
  }

  const orchestraRoot = options.orchestraRoot || requireOrchestraRoot();
  const config = loadConfig(orchestraRoot);
  const paths = getResolvedPaths(orchestraRoot, config);

  // Load manifest and progress
  const manifestResult = loadManifest(paths.manifest);
  if (!manifestResult.success || !manifestResult.data) {
    throw new OrchestraError(ErrorCode.CONFIG_ERROR, "Cannot load manifest");
  }

  const progressResult = loadProgress(paths.progress);
  if (!progressResult.success || !progressResult.data) {
    throw new OrchestraError(ErrorCode.CONFIG_ERROR, "Cannot load progress");
  }

  // Determine task ID
  const taskId = options.task
    ? parseInt(options.task, 10)
    : progressResult.data.currentTask;

  if (!taskId) {
    throw new OrchestraError(ErrorCode.VALIDATION_ERROR, "No task specified");
  }

  const task = getTask(manifestResult.data, taskId);
  if (!task) {
    throw new OrchestraError(
      ErrorCode.VALIDATION_ERROR,
      `Task ${taskId} not found`
    );
  }

  // Check current status
  const taskProgress = progressResult.data.tasks?.find((t) => t.id === taskId);
  const previousStatus = taskProgress?.status || task.status || "unknown";

  if (previousStatus === "COMPLETE" || previousStatus === "completed") {
    throw new OrchestraError(
      ErrorCode.VALIDATION_ERROR,
      `Task ${taskId} is already complete`
    );
  }

  if (previousStatus === "ESCALATED") {
    // Already escalated - just return info
    return {
      success: true,
      taskId,
      previousStatus: "ESCALATED",
      newStatus: "ESCALATED",
      reason: options.reason,
      context: options.context,
      escalatedAt: taskProgress?.escalatedAt || new Date().toISOString(),
      reportPath: path.join(paths.artifacts, `task-${taskId}`, "escalation-report.md"),
      attemptHistory: [],
      humanOptions: ["fix_manually", "modify_spec", "skip_task", "abort_sprint"],
    };
  }

  // Compile attempt history
  const attemptHistory = compileAttemptHistory(paths.feedback, taskId);

  // Find feedback files
  const feedbackFiles: string[] = [];
  if (fs.existsSync(paths.feedback)) {
    feedbackFiles.push(
      ...fs.readdirSync(paths.feedback)
        .filter((f) => f.startsWith(`task-${taskId}-`))
        .map((f) => path.join(paths.feedback, f))
    );
  }

  // Generate report
  const report = generateEscalationReport(
    task,
    options.reason,
    options.context,
    attemptHistory,
    feedbackFiles
  );

  // Write report
  const artifactDir = path.join(paths.artifacts, `task-${taskId}`);
  fs.mkdirSync(artifactDir, { recursive: true });
  const reportPath = path.join(artifactDir, "escalation-report.md");
  fs.writeFileSync(reportPath, report);

  // Update progress to ESCALATED
  const escalatedAt = new Date().toISOString();
  // (implement updateProgress with ESCALATED status)

  return {
    success: true,
    taskId,
    previousStatus,
    newStatus: "ESCALATED",
    reason: options.reason,
    context: options.context,
    escalatedAt,
    reportPath,
    attemptHistory,
    humanOptions: ["fix_manually", "modify_spec", "skip_task", "abort_sprint"],
  };
}

### CLI Command: src/commands/escalate.ts

```typescript
import { Command } from "commander";
import { runEscalate } from "../core/escalate.js";
import * as output from "../core/output.js";

export function createEscalateCommand(): Command {
  const cmd = new Command("escalate")
    .description("Escalate persistent failures to human supervisor")
    .requiredOption("-r, --reason <reason>", "Why escalation is needed")
    .option("-t, --task <id>", "Task ID (defaults to current)")
    .option("-c, --context <text>", "Additional context")
    .option("--json", "Output as JSON")
    .action(async (options) => {
      try {
        const result = await runEscalate({
          task: options.task,
          reason: options.reason,
          context: options.context,
        });

        if (options.json) {
          console.log(JSON.stringify(result, null, 2));
        } else {
          output.success(`Task ${result.taskId} escalated to human supervisor`);
          console.log(`\nReason: ${result.reason}`);
          console.log(`Status: ${result.newStatus}`);
          console.log(`\nEscalation report: ${result.reportPath}`);
          
          output.warn("\nWorkflow halted. Human intervention required.");
          
          console.log("\nHuman options:");
          console.log(`  - Fix manually and run: orchestra complete --task ${result.taskId} --force`);
          console.log(`  - Modify spec and retry: Edit manifest, then orchestra prepare --task ${result.taskId}`);
          console.log(`  - Skip this task: orchestra complete --task ${result.taskId} --skip --reason "..."`);
          console.log(`  - Abort sprint: orchestra status --abort --reason "..."`);
        }
      } catch (error) {
        output.error(error instanceof Error ? error.message : String(error));
        process.exit(1);
      }
    });

  return cmd;
}
```
