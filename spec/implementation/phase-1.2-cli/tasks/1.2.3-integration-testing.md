### 3. Feedback Privacy

```typescript
describe('Feedback Privacy', () => {
  it('does not reveal verification criteria in feedback', async () => {
    // Setup: verification with hidden criteria
    const verifyResult = {
      checks: [
        {
          id: 'check-1',
          type: 'content_contains',
          path: 'src/secret.ts',
          pattern: 'hidden-pattern',
          passed: false,
          description: 'Function handles errors correctly',
          guidance: 'Add error handling'
        }
      ]
    };
    
    // Run feedback
    const result = await runFeedback({ verificationResult });
    
    // Read feedback file
    const content = fs.readFileSync(result.feedbackPath, 'utf-8');
    
    // Assert: hidden details not present
    expect(content).not.toContain('content_contains');
    expect(content).not.toContain('src/secret.ts');
    expect(content).not.toContain('hidden-pattern');
    
    // Assert: friendly description IS present
    expect(content).toContain('Function handles errors correctly');
    expect(content).toContain('Add error handling');
  });
});
```

### 4. Complete Workflow

```typescript
describe('Complete Failure Workflow', () => {
  it('runs full cycle: verify → feedback → retry → feedback → retry → feedback → escalate', async () => {
    // This is the "3 strikes" scenario
    
    for (let attempt = 1; attempt <= 3; attempt++) {
      // Verify fails
      const verifyResult = await runVerification({ task: '1' });
      expect(verifyResult.passed).toBe(false);
      
      // Generate feedback
      const feedbackResult = await runFeedback({ task: '1' });
      expect(feedbackResult.attempt).toBe(attempt);
      expect(feedbackResult.canRetry).toBe(attempt < 3);
    }
    
    // Now escalate
    const escalateResult = await runEscalate({
      task: '1',
      reason: 'Max attempts exceeded'
    });
    
    expect(escalateResult.newStatus).toBe('ESCALATED');
    expect(escalateResult.attemptHistory).toHaveLength(3);
  });
});
```

### 5. Edge Cases

```typescript
describe('Edge Cases', () => {
  it('handles feedback when no verification run', async () => {
    await expect(runFeedback({ task: '1' }))
      .rejects.toThrow('No verification results');
  });
  
  it('handles escalate without reason', async () => {
    await expect(runEscalate({ task: '1', reason: '' }))
      .rejects.toThrow('--reason is required');
  });
  
  it('handles already escalated task', async () => {
    // First escalation
    await runEscalate({ task: '1', reason: 'First' });
    
    // Second escalation - should not error
    const result = await runEscalate({ task: '1', reason: 'Second' });
    expect(result.previousStatus).toBe('ESCALATED');
  });
  
  it('prevents feedback on completed task', async () => {
    // Complete the task
    await runComplete({ task: '1' });
    
    // Try feedback
    await expect(runFeedback({ task: '1' }))
      .rejects.toThrow('already complete');
  });
});
```
