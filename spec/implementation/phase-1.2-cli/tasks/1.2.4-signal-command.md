```typescript
/**
 * Signal Creation - Bible Section 8.3
 * 
 * Implementor signals task completion.
 * Creates signal file for orchestrator to validate.
 */

import * as fs from "node:fs";
import * as path from "node:path";
import { loadManifest, getTask } from "./manifest.js";
import { loadProgress } from "./progress.js";
import { loadConfig, getResolvedPaths, requireOrchestraRoot } from "./config.js";
import { getChangedFiles } from "./git.js";
import { OrchestraError, ErrorCode } from "./errors.js";
import * as yaml from "js-yaml";

export interface SignalOptions {
  task?: string;
  summary: string;
  files?: string[];
  tests?: string[];
  notes?: string;
  orchestraRoot?: string;
}

export interface SignalResult {
  success: boolean;
  taskId: number;
  summary: string;
  signalPath: string;
  artifacts: {
    created: string[];
    modified: string[];
  };
  signaledAt: string;
  nextStep: string;
}

export async function runSignal(options: SignalOptions): Promise<SignalResult> {
  if (!options.summary) {
    throw new OrchestraError(ErrorCode.VALIDATION_ERROR, "--summary is required");
  }

  const orchestraRoot = options.orchestraRoot || requireOrchestraRoot();
  const config = loadConfig(orchestraRoot);
  const paths = getResolvedPaths(orchestraRoot, config);

  // Load manifest and progress
  const manifestResult = loadManifest(paths.manifest);
  if (!manifestResult.success || !manifestResult.data) {
    throw new OrchestraError(ErrorCode.CONFIG_ERROR, "Cannot load manifest");
  }

  const progressResult = loadProgress(paths.progress);
  if (!progressResult.success || !progressResult.data) {
    throw new OrchestraError(ErrorCode.CONFIG_ERROR, "Cannot load progress");
  }

  // Determine task ID
  const taskId = options.task
    ? parseInt(options.task, 10)
    : progressResult.data.currentTask;

  if (!taskId) {
    throw new OrchestraError(ErrorCode.VALIDATION_ERROR, "No task specified");
  }

  const task = getTask(manifestResult.data, taskId);
  if (!task) {
    throw new OrchestraError(
      ErrorCode.VALIDATION_ERROR,
      `Task ${taskId} not found`
    );
  }

  // Verify task is in progress
  const taskProgress = progressResult.data.tasks?.find((t) => t.id === taskId);
  const status = taskProgress?.status || task.status;
  
  if (status !== "in_progress") {
    throw new OrchestraError(
      ErrorCode.VALIDATION_ERROR,
      `Task ${taskId} is not in progress. Current status: ${status}`
    );
  }

  // Verify handover exists
  const handoverPath = path.join(paths.handovers, "current-task.md");
  if (!fs.existsSync(handoverPath)) {
    throw new OrchestraError(
      ErrorCode.VALIDATION_ERROR,
      `No handover found for task ${taskId}. The orchestrator must prepare the task first.`
    );
  }

  // Get or detect artifacts
  let created: string[] = [];
  let modified: string[] = [];
  
  if (options.files && options.files.length > 0) {
    // User specified - assume all are relevant
    modified = options.files;
  } else {
    // Auto-detect from git
    try {
      const changes = await getChangedFiles();
      created = changes.filter((c) => c.status === "added").map((c) => c.path);
      modified = changes.filter((c) => c.status === "modified").map((c) => c.path);
    } catch {
      // Git not available or not a repo - continue without auto-detect
    }
  }

  const tests = options.tests || [];

  // Create signal file
  const signaledAt = new Date().toISOString();
  const signalData = {
    version: "1.0",
    task_id: taskId,
    signaled_at: signaledAt,
    summary: options.summary,
    artifacts: {
      created,
      modified,
    },
    tests,
    notes: options.notes || null,
  };

  const signalsDir = path.join(paths.signals || path.join(orchestraRoot, "implementor", "signals"));
  fs.mkdirSync(signalsDir, { recursive: true });
  
  const signalPath = path.join(signalsDir, `task-${taskId}-signal.yaml`);
  fs.writeFileSync(signalPath, yaml.dump(signalData));

  return {
    success: true,
    taskId,
    summary: options.summary,
    signalPath,
    artifacts: { created, modified },
    signaledAt,
    nextStep: "Orchestrator runs accept-signal",
  };
}
```

### CLI Command: src/commands/signal.ts

```typescript
import { Command } from "commander";
import { runSignal } from "../core/signal.js";
import * as output from "../core/output.js";

export function createSignalCommand(): Command {
  const cmd = new Command("signal")
    .description("Signal that a task is complete (Implementor)")
    .requiredOption("-s, --summary <text>", "Brief description of what was done")
    .option("-t, --task <id>", "Task ID (defaults to current)")
    .option("-f, --files <files...>", "Files created or modified")
    .option("--tests <tests...>", "Test files added")
    .option("-n, --notes <text>", "Additional notes")
    .option("--json", "Output as JSON")
    .action(async (options) => {
      try {
        const result = await runSignal({
          task: options.task,
          summary: options.summary,
          files: options.files,
          tests: options.tests,
          notes: options.notes,
        });

        if (options.json) {
          console.log(JSON.stringify(result, null, 2));
        } else {
          output.success(`Signal created for task ${result.taskId}`);
          console.log(`\nSummary: ${result.summary}`);
          
          if (result.artifacts.created.length > 0 || result.artifacts.modified.length > 0) {
            console.log("\nArtifacts:");
            if (result.artifacts.created.length > 0) {
              console.log(`  Created: ${result.artifacts.created.join(", ")}`);
            }
            if (result.artifacts.modified.length > 0) {
              console.log(`  Modified: ${result.artifacts.modified.join(", ")}`);
            }
          }
          
          console.log(`\nSignal file: ${result.signalPath}`);
          console.log("\nNext: Orchestrator will run 'orchestra accept-signal' to validate");
        }
      } catch (error) {
        output.error(error instanceof Error ? error.message : String(error));
        process.exit(1);
      }
    });

  return cmd;
}
```
