## Implementation

```typescript
// src/mcp/tools/complete_task.ts
import { Tool } from '@modelcontextprotocol/sdk/types.js';
import { registerTool } from './registry.js';
import { loadConfig, getResolvedPaths, requireOrchestraRoot } from '../../core/config.js';
import { loadManifest, updateTaskStatus, getTask, getTasksByStatus } from '../../core/manifest.js';
import { OrchestraMcpError, ErrorCode } from '../errors.js';
import * as fs from 'node:fs';
import * as path from 'node:path';

const definition: Tool = {
  name: 'complete_task',
  description: 'Mark the current task as complete and prepare for next task.',
  inputSchema: {
    type: 'object',
    properties: {
      force: {
        type: 'boolean',
        description: 'Force completion without verifying signal acceptance',
      },
    },
  },
};

async function handler(params: Record<string, unknown>) {
  const force = Boolean(params.force);
  const orchestraRoot = requireOrchestraRoot();
  const config = loadConfig(orchestraRoot);
  const paths = getResolvedPaths(orchestraRoot, config);
  
  const actions: string[] = [];
  
  // Load manifest
  const manifestResult = loadManifest(paths.manifest);
  if (!manifestResult.success || !manifestResult.data) {
    throw new OrchestraMcpError(
      ErrorCode.VALIDATION_ERROR,
      'Could not load manifest',
      manifestResult.error
    );
  }
  const manifest = manifestResult.data;
  
  // Find current task in IMPLEMENT status
  const implementTasks = getTasksByStatus(manifest, 'IMPLEMENT');
  if (implementTasks.length === 0) {
    throw new OrchestraMcpError(
      ErrorCode.WORKFLOW_ERROR,
      'No task currently in IMPLEMENT status'
    );
  }
  const task = implementTasks[0];
  const taskId = task.id;
  
  // Check for accepted signal (unless force)
  const signalPath = path.join(paths.handovers, 'completion-signal.md');
  const progressPath = path.join(paths.handovers, 'progress.yaml');
  
  if (!force) {
    // Verify completion signal was accepted
    if (!fs.existsSync(progressPath)) {
      throw new OrchestraMcpError(
        ErrorCode.WORKFLOW_ERROR,
        'No progress.yaml found - signal not processed yet',
        'Run signal_complete and accept_signal first'
      );
    }
    
    const progressContent = fs.readFileSync(progressPath, 'utf-8');
    if (!progressContent.includes('status: PASSED') && !progressContent.includes('status: "PASSED"')) {
      throw new OrchestraMcpError(
        ErrorCode.WORKFLOW_ERROR,
        'Completion signal not accepted',
        'Run accept_signal first or use force=true'
      );
    }
    actions.push('Verified completion signal accepted');
  } else {
    actions.push('Skipped signal verification (force=true)');
  }
  
  // Update manifest status
  const previousStatus = task.status;
  const updateResult = updateTaskStatus(manifest, taskId, 'COMPLETE');
  if (!updateResult.success) {
    throw new OrchestraMcpError(
      ErrorCode.UPDATE_ERROR,
      `Failed to update task status: ${updateResult.error}`
    );
  }
  actions.push(`Updated task ${taskId} status: ${previousStatus} → COMPLETE`);
  
  // Save updated manifest
  fs.writeFileSync(paths.manifest, JSON.stringify(manifest, null, 2));
  actions.push('Saved manifest.yaml');
  
  // Archive handover files
  const archiveDir = path.join(paths.handovers, '..', 'archive', `task-${taskId}`);
  fs.mkdirSync(archiveDir, { recursive: true });
  
  const filesToArchive = [
    'current-task.md',
    'completion-signal.md',
    'progress.yaml',
  ];
  
  for (const file of filesToArchive) {
    const srcPath = path.join(paths.handovers, file);
    if (fs.existsSync(srcPath)) {
      const destPath = path.join(archiveDir, file);
      fs.copyFileSync(srcPath, destPath);
      fs.unlinkSync(srcPath);
    }
  }
  actions.push(`Archived handover files to ${archiveDir}`);
  
  // Find next pending task
  const pendingTasks = getTasksByStatus(manifest, 'PENDING');
  const nextTask = pendingTasks.length > 0
    ? {
        id: pendingTasks[0].id,
        title: pendingTasks[0].title,
        status: pendingTasks[0].status,
      }
    : null;
  
  if (nextTask) {
    actions.push(`Next pending task: ${nextTask.id} - ${nextTask.title}`);
  } else {
    actions.push('No more pending tasks - phase complete!');
  }
  
  return {
    success: true,
    taskId,
    previousStatus,
    newStatus: 'COMPLETE',
    actions,
    archivePath: archiveDir,
    nextTask,
  };
}

// Register the tool
registerTool(definition, handler);

export { definition, handler };
```

## Test Cases

```typescript
describe('complete_task tool', () => {
  it('completes task when signal accepted', async () => {
    // Setup: task in IMPLEMENT, signal accepted
    const result = await handler({});
    
    expect(result.success).toBe(true);
    expect(result.newStatus).toBe('COMPLETE');
    expect(result.archivePath).toBeDefined();
  });
  
  it('fails without accepted signal', async () => {
    // Setup: task in IMPLEMENT, no signal
    await expect(handler({})).rejects.toThrow('signal not processed');
  });
  
  it('succeeds with force flag', async () => {
    // Setup: task in IMPLEMENT, no signal
    const result = await handler({ force: true });
    
    expect(result.success).toBe(true);
    expect(result.actions).toContain('Skipped signal verification (force=true)');
  });
  
  it('archives handover files', async () => {
    const result = await handler({ force: true });
    
    expect(fs.existsSync(result.archivePath)).toBe(true);
    expect(fs.existsSync(path.join(result.archivePath, 'current-task.md'))).toBe(true);
  });
  
  it('identifies next pending task', async () => {
    const result = await handler({ force: true });
    
    expect(result.nextTask).toBeDefined();
    expect(result.nextTask?.status).toBe('PENDING');
  });
});
```
