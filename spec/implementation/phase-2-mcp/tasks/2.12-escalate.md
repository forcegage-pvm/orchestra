## Implementation

```typescript
import { runEscalate } from '../../core/index.js';
import { ToolHandler } from '../types.js';

export const escalateTool: ToolHandler = {
  name: 'escalate',
  description: 'Escalate persistent failures to human supervisor (Orchestrator)',
  inputSchema: {
    type: 'object',
    properties: {
      taskId: {
        type: 'number',
        description: 'Task ID to escalate'
      },
      reason: {
        type: 'string',
        description: 'Why escalation is needed'
      },
      attempts: {
        type: 'number',
        description: 'Number of attempts made'
      },
      context: {
        type: 'string',
        description: 'Additional context for human'
      }
    },
    required: ['reason']
  },
  handler: async (params) => {
    const result = await runEscalate({
      task: params.taskId?.toString(),
      reason: params.reason,
      context: params.context
    });
    return {
      success: result.success,
      taskId: result.taskId,
      previousStatus: result.previousStatus,
      newStatus: 'escalated',
      escalationPath: result.escalationPath,
      reportPath: result.reportPath
    };
  }
};
```
