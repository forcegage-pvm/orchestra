```typescript
// test/mcp/integration/full-workflow.test.ts
import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import { spawn, ChildProcess } from 'node:child_process';
import { Client } from '@modelcontextprotocol/sdk/client/index.js';
import { StdioClientTransport } from '@modelcontextprotocol/sdk/client/stdio.js';
import * as fs from 'node:fs';
import * as path from 'node:path';

describe('MCP Server Integration', () => {
  let client: Client;
  let serverProcess: ChildProcess;
  let testDir: string;
  
  beforeEach(async () => {
    // Create temp test directory
    testDir = fs.mkdtempSync(path.join(os.tmpdir(), 'orchestra-mcp-'));
    
    // Setup orchestra structure
    setupTestOrchestra(testDir);
    
    // Start MCP server
    serverProcess = spawn('node', ['dist/mcp/server.js'], {
      cwd: testDir,
      env: { ...process.env, ORCHESTRA_ROOT: testDir },
    });
    
    // Connect client
    const transport = new StdioClientTransport({
      command: 'node',
      args: ['dist/mcp/server.js'],
      cwd: testDir,
    });
    
    client = new Client({ name: 'test-client', version: '1.0.0' }, {});
    await client.connect(transport);
  });
  
  afterEach(async () => {
    await client.close();
    serverProcess.kill();
    fs.rmSync(testDir, { recursive: true, force: true });
  });
  
  describe('Complete Task Lifecycle', () => {
    it('executes full workflow: prepare → signal → accept_signal → verify → complete', async () => {
      // 1. Get status - should show task 1 pending
      const status1 = await client.callTool('status', {});
      expect(status1.content[0].text).toContain('"currentTask":null');
      
      // 2. Prepare task 1 (Orchestrator)
      const prepare = await client.callTool('prepare', { taskId: 1 });
      expect(prepare.content[0].text).toContain('"success":true');
      
      // 3. Get status - should show task 1 in progress
      const status2 = await client.callTool('status', {});
      expect(status2.content[0].text).toContain('"status":"in_progress"');  // lowercase
      
      // 4. Signal completion (Implementor)
      const signal = await client.callTool('signal', { 
        summary: 'Implemented feature X' 
      });
      expect(signal.content[0].text).toContain('"success":true');
      
      // 5. Accept signal (Orchestrator)
      const accept = await client.callTool('accept_signal', { taskId: 1 });
      expect(accept.content[0].text).toContain('"accepted":true');
      
      // 6. Verify (Orchestrator - hidden criteria)
      const verify = await client.callTool('verify', { taskId: 1 });
      expect(verify.content[0].text).toContain('"passed":true');
      
      // 7. Complete (Orchestrator)
      const complete = await client.callTool('complete', { taskId: 1 });
      expect(complete.content[0].text).toContain('"newStatus":"completed"');
      expect(context.content[0].text).toContain('"taskId":1');
      expect(context.content[0].text).toContain('handover');
      
      // 6. Signal completion
      const signal = await client.callTool('signal_complete', {
        summary: 'Implemented the feature',
        filesChanged: ['src/feature.ts'],
        testsAdded: ['test/feature.test.ts'],
        verification: 'All tests pass',
      });
      expect(signal.content[0].text).toContain('"success":true');
      
      // 7. Complete task
      const complete = await client.callTool('complete_task', { force: true });
      expect(complete.content[0].text).toContain('"success":true');
      expect(complete.content[0].text).toContain('"newStatus":"COMPLETE"');
      
      // 8. Get final status
      const status3 = await client.callTool('get_status', {});
      expect(status3.content[0].text).toContain('"completed":1');
      expect(status3.content[0].text).toContain('"pending":2');
    });
  });
});
```

```typescript
// test/mcp/integration/error-handling.test.ts
describe('Error Handling', () => {
  it('returns structured error when manifest missing', async () => {
    // Remove manifest
    fs.unlinkSync(path.join(testDir, '.orchestra', 'manifest.yaml'));
    
    const result = await client.callTool('get_status', {});
    expect(result.isError).toBe(true);
    expect(result.content[0].text).toContain('CONFIG_ERROR');
  });
  
  it('rejects prepare when task already in progress', async () => {
    await client.callTool('prepare_task', { taskId: 1 });
    
    const result = await client.callTool('prepare_task', { taskId: 2 });
    expect(result.isError).toBe(true);
    expect(result.content[0].text).toContain('WORKFLOW_ERROR');
  });
  
  it('rejects complete without signal accepted', async () => {
    await client.callTool('prepare_task', { taskId: 1 });
    
    const result = await client.callTool('complete_task', {});
    expect(result.isError).toBe(true);
    expect(result.content[0].text).toContain('signal not processed');
  });
  
  it('handles invalid task ID gracefully', async () => {
    const result = await client.callTool('prepare_task', { taskId: 999 });
    expect(result.isError).toBe(true);
    expect(result.content[0].text).toContain('not found');
  });
});
```

### Edge Cases Tests

```typescript
// test/mcp/integration/edge-cases.test.ts
describe('Edge Cases', () => {
  it('handles concurrent prepare requests', async () => {
    // Send two prepare requests simultaneously
    const [result1, result2] = await Promise.all([
      client.callTool('prepare_task', { taskId: 1 }),
      client.callTool('prepare_task', { taskId: 1 }),
    ]);
    
    // One should succeed, one should fail
    const successes = [result1, result2].filter(r => !r.isError);
    expect(successes).toHaveLength(1);
  });
  
  it('recovers from corrupted progress.yaml', async () => {
    await client.callTool('prepare_task', { taskId: 1 });
    
    // Corrupt the progress file
    fs.writeFileSync(
      path.join(testDir, '.orchestra', 'handovers', 'progress.yaml'),
      'invalid: [yaml: content'
    );
    
    // Should still be able to get status
    const result = await client.callTool('get_status', {});
    expect(result.isError).toBe(false);
  });
  
  it('handles empty manifest tasks array', async () => {
    // Create manifest with no tasks
    const manifest = { meta: { version: '1.0.0' }, tasks: [] };
    fs.writeFileSync(
      path.join(testDir, '.orchestra', 'manifest.yaml'),
      JSON.stringify(manifest)
    );
    
    const result = await client.callTool('get_status', {});
    expect(result.content[0].text).toContain('"total":0');
  });
  
  it('handles very long task descriptions', async () => {
    const longDesc = 'A'.repeat(10000);
    const result = await client.callTool('signal_complete', {
      summary: longDesc,
      filesChanged: [],
    });
    
    expect(result.isError).toBe(false);
  });
});
```
