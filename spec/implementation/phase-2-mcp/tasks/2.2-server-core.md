## Implementation

### server.ts

```typescript
import { Server } from '@modelcontextprotocol/sdk/server/index.js';
import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js';
import {
  CallToolRequestSchema,
  ListToolsRequestSchema,
} from '@modelcontextprotocol/sdk/types.js';
import { registerTools, getToolHandler } from './tools/registry.js';
import { OrchestraMcpError, ErrorCode } from './errors.js';

const server = new Server(
  {
    name: 'orchestra',
    version: '1.0.0',
  },
  {
    capabilities: {
      tools: {},
    },
  }
);

// Register tool listing
server.setRequestHandler(ListToolsRequestSchema, async () => {
  return {
    tools: registerTools(),
  };
});

// Register tool execution
server.setRequestHandler(CallToolRequestSchema, async (request) => {
  const { name, arguments: args } = request.params;
  
  const handler = getToolHandler(name);
  if (!handler) {
    throw new OrchestraMcpError(
      ErrorCode.MethodNotFound,
      `Unknown tool: ${name}`
    );
  }
  
  try {
    const result = await handler(args);
    return {
      content: [
        {
          type: 'text',
          text: JSON.stringify(result, null, 2),
        },
      ],
    };
  } catch (error) {
    if (error instanceof OrchestraMcpError) {
      throw error;
    }
    throw new OrchestraMcpError(
      ErrorCode.ApplicationError,
      error instanceof Error ? error.message : 'Unknown error'
    );
  }
});

async function main() {
  const transport = new StdioServerTransport();
  
  // Graceful shutdown
  process.on('SIGINT', async () => {
    await server.close();
    process.exit(0);
  });
  
  await server.connect(transport);
  console.error('Orchestra MCP server running on stdio');
}

main().catch((error) => {
  console.error('Fatal error:', error);
  process.exit(1);
});
```

### errors.ts

```typescript
export enum ErrorCode {
  // MCP standard errors
  ParseError = -32700,
  InvalidRequest = -32600,
  MethodNotFound = -32601,
  InvalidParams = -32602,
  InternalError = -32603,
  
  // Orchestra-specific errors (-32000 to -32099)
  ApplicationError = -32000,
  ConfigurationError = -32001,
  StateError = -32002,
  TaskNotFound = -32003,
  ValidationError = -32004,
}

export class OrchestraMcpError extends Error {
  constructor(
    public readonly code: ErrorCode,
    message: string,
    public readonly data?: unknown
  ) {
    super(message);
    this.name = 'OrchestraMcpError';
  }
  
  toJSON() {
    return {
      code: this.code,
      message: this.message,
      data: this.data,
    };
  }
}
```

### tools/registry.ts

```typescript
import type { Tool } from '@modelcontextprotocol/sdk/types.js';

export type ToolHandler = (args: Record<string, unknown>) => Promise<unknown>;

interface RegisteredTool {
  definition: Tool;
  handler: ToolHandler;
}

const tools: Map<string, RegisteredTool> = new Map();

export function registerTool(
  definition: Tool,
  handler: ToolHandler
): void {
  tools.set(definition.name, { definition, handler });
}

export function registerTools(): Tool[] {
  return Array.from(tools.values()).map((t) => t.definition);
}

export function getToolHandler(name: string): ToolHandler | undefined {
  return tools.get(name)?.handler;
}
```
