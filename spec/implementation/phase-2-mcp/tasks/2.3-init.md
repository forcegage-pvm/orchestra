## Implementation

```typescript
import { runInit } from '../../core/index.js';
import { ToolHandler } from '../types.js';

export const initTool: ToolHandler = {
  name: 'init',
  description: 'Initialize a new sprint from specification (Orchestrator)',
  inputSchema: {
    type: 'object',
    properties: {
      specPath: {
        type: 'string',
        description: 'Path to sprint specification file'
      }
    },
    required: ['specPath']
  },
  handler: async (params) => {
    const result = await runInit({ spec: params.specPath });
    return {
      success: result.success,
      sprintId: result.sprintId,
      taskCount: result.taskCount,
      orchestraRoot: result.orchestraRoot
    };
  }
};
```
