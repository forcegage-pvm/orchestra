## Implementation

```typescript
// src/mcp/tools/signal_complete.ts
import { Tool } from '@modelcontextprotocol/sdk/types.js';
import { registerTool } from './registry.js';
import { runAcceptSignal } from '../../core/signal.js';
import { runVerification } from '../../core/verification.js';
import { requireOrchestraRoot } from '../../core/config.js';
import { OrchestraMcpError, ErrorCode } from '../errors.js';
import * as fs from 'node:fs';
import * as path from 'node:path';

const definition: Tool = {
  name: 'signal_complete',
  description: 'Signal that the current task is complete and ready for verification.',
  inputSchema: {
    type: 'object',
    properties: {
      summary: {
        type: 'string',
        description: 'Brief description of what was implemented',
      },
      artifactsCreated: {
        type: 'array',
        items: { type: 'string' },
        description: 'List of files created or modified',
      },
      testsAdded: {
        type: 'array',
        items: { type: 'string' },
        description: 'List of test files added',
      },
      notes: {
        type: 'string',
        description: 'Optional additional notes',
      },
      skipVerify: {
        type: 'boolean',
        description: 'Skip verification after signal',
      },
    },
    required: ['summary', 'artifactsCreated'],
  },
};

interface SignalCompleteParams {
  summary: string;
  artifactsCreated: string[];
  testsAdded?: string[];
  notes?: string;
  skipVerify?: boolean;
}

async function handler(params: SignalCompleteParams) {
  const orchestraRoot = requireOrchestraRoot();
  
  // 1. Write completion signal file (implementor's responsibility)
  await writeCompletionSignal(orchestraRoot, params);
  
  // 2. Run accept-signal check
  const signalResult = await runAcceptSignal({});
  
  const response: any = {
    accepted: signalResult.overall === 'ACCEPTED',
    taskId: signalResult.taskId,
    signalChecks: signalResult.checks.map((c) => ({
      id: c.id,
      check: c.check,
      passed: c.passed,
      message: c.message,
    })),
    verification: undefined,
    nextStep: 'fix_signal',
    actionRequired: undefined,
  };
  
  // If signal not accepted, return early
  if (!response.accepted) {
    response.actionRequired = signalResult.checks
      .filter((c) => !c.passed)
      .map((c) => c.fix)
      .join('; ');
    return response;
  }
  
  // 3. Optionally run verification
  if (!params.skipVerify) {
    const verifyResult = await runVerification({ skipAccept: true });
    
    response.verification = {
      ran: true,
      passed: verifyResult.report.overallPassed,
      checks: verifyResult.report.results.map((r) => ({
        id: r.checkId,
        description: r.description,
        passed: r.passed,
        severity: r.severity,
      })),
    };
    
    if (verifyResult.report.overallPassed) {
      response.nextStep = 'complete';
      response.actionRequired = 'Run "orchestra complete" to finish the task';
    } else {
      response.nextStep = 'fix_verification';
      const failedChecks = verifyResult.report.results.filter((r) => !r.passed);
      response.actionRequired = `Fix ${failedChecks.length} failing check(s): ${
        failedChecks.map((c) => c.checkId).join(', ')
      }`;
    }
  } else {
    response.nextStep = 'complete';
    response.actionRequired = 'Run verification with "orchestra verify"';
  }
  
  return response;
}

async function writeCompletionSignal(
  orchestraRoot: string,
  params: SignalCompleteParams
): Promise<void> {
  const signalPath = path.join(
    orchestraRoot,
    '.orchestra',
    'handover',
    'completion-signal.md'
  );
  
  const content = `# Completion Signal

## Summary
${params.summary}

## Artifacts Created
${params.artifactsCreated.map((a) => `- ${a}`).join('\n')}

## Tests
${params.testsAdded?.length
  ? params.testsAdded.map((t) => `- ${t}`).join('\n')
  : '- N/A'}

${params.notes ? `## Notes\n${params.notes}\n` : ''}
`;
  
  fs.mkdirSync(path.dirname(signalPath), { recursive: true });
  fs.writeFileSync(signalPath, content, 'utf-8');
}

// Register the tool
registerTool(definition, handler as any);

export { definition, handler };
```

## Test Cases

```typescript
describe('signal_complete tool', () => {
  beforeEach(() => {
    // Setup: prepare a task first
  });
  
  it('accepts signal with valid params', async () => {
    const result = await handler({
      summary: 'Implemented the widget',
      artifactsCreated: ['src/widget.ts'],
      testsAdded: ['test/widget.test.ts'],
    });
    
    expect(result.accepted).toBe(true);
    expect(result.signalChecks).toBeDefined();
  });
  
  it('rejects signal without pre-signal artifact', async () => {
    const result = await handler({
      summary: 'Did some work',
      artifactsCreated: ['file.ts'],
    });
    
    expect(result.accepted).toBe(false);
    expect(result.nextStep).toBe('fix_signal');
  });
  
  it('runs verification when skipVerify is false', async () => {
    const result = await handler({
      summary: 'Complete implementation',
      artifactsCreated: ['src/main.ts'],
      skipVerify: false,
    });
    
    expect(result.verification?.ran).toBe(true);
  });
  
  it('skips verification when skipVerify is true', async () => {
    const result = await handler({
      summary: 'Complete implementation',
      artifactsCreated: ['src/main.ts'],
      skipVerify: true,
    });
    
    expect(result.verification).toBeUndefined();
  });
});
```

## Notes

### Pre-Signal Requirement

The implementor MUST still run pre-signal checks before calling this tool:

```powershell
.\.orchestra\implementor\.implementor-only\scripts\pre-signal-check.ps1
```

The tool checks for the pre-signal artifact and will reject the signal if it's missing.

### Verification Flow

```
signal_complete called
    │
    ├─► Write completion-signal.md
    │
    ├─► Run accept-signal checks
    │   ├─► Pre-signal artifact exists?
    │   ├─► Pre-signal status PASSED?
    │   ├─► Task ID matches?
    │   ├─► Artifact fresh?
    │   ├─► Completion signal filled?
    │   └─► Deliverables check passed?
    │
    ├─► If all pass & !skipVerify → Run verification
    │
    └─► Return result with next steps
```
