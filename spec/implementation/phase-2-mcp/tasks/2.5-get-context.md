## Implementation

```typescript
// src/mcp/tools/get_context.ts
import { Tool } from '@modelcontextprotocol/sdk/types.js';
import { registerTool } from './registry.js';
import { loadManifest, getTask, getTasksByStatus } from '../../core/manifest.js';
import { loadProgress, getProgressSummary } from '../../core/progress.js';
import { loadConfig, getResolvedPaths, requireOrchestraRoot } from '../../core/config.js';
import { OrchestraMcpError, ErrorCode } from '../errors.js';
import * as fs from 'node:fs';
import * as path from 'node:path';

const definition: Tool = {
  name: 'get_context',
  description: 'Get the current task context and sprint status.',
  inputSchema: {
    type: 'object',
    properties: {},
  },
};

async function handler(_params: Record<string, unknown>) {
  const orchestraRoot = requireOrchestraRoot();
  const config = loadConfig(orchestraRoot);
  const paths = getResolvedPaths(orchestraRoot, config);
  
  // Load manifest
  const manifestResult = loadManifest(paths.manifest);
  if (!manifestResult.success || !manifestResult.data) {
    throw new OrchestraMcpError(
      ErrorCode.StateError,
      'Failed to load manifest: ' + manifestResult.message
    );
  }
  
  const manifest = manifestResult.data;
  
  // Load progress
  const progress = loadProgress(manifest.sprint.id, orchestraRoot);
  const progressSummary = getProgressSummary(progress);
  
  // Get current task (in IMPLEMENT status)
  const implementingTasks = getTasksByStatus(manifest, 'IMPLEMENT');
  const currentTask = implementingTasks.length > 0
    ? implementingTasks[0]
    : null;
  
  // Calculate progress stats
  const completedTasks = getTasksByStatus(manifest, 'COMPLETE');
  const pendingTasks = getTasksByStatus(manifest, 'PENDING');
  const totalTasks = manifest.tasks.length;
  
  // Check handover files
  const handoverPath = paths.handovers;
  const currentTaskPath = path.join(handoverPath, 'current-task.md');
  const completionSignalPath = path.join(handoverPath, 'completion-signal.md');
  
  const handoverExists = fs.existsSync(currentTaskPath);
  
  // Get recent activity (last 5 entries)
  const recentActivity = progress.entries
    .slice(-5)
    .reverse()
    .map((entry) => ({
      taskId: entry.task_id,
      status: entry.status,
      timestamp: entry.timestamp,
    }));
  
  return {
    sprint: {
      id: manifest.sprint.id,
      name: manifest.sprint.name,
      status: manifest.sprint.status,
      version: manifest.version,
    },
    currentTask: currentTask
      ? {
          id: currentTask.id,
          title: currentTask.title,
          description: currentTask.description ?? '',
          status: currentTask.status,
          category: currentTask.category ?? 'INTEGRATION',
          dependencies: currentTask.dependencies ?? [],
          specRef: currentTask.speckit_task_ref?.[0],
        }
      : null,
    progress: {
      completed: completedTasks.length,
      inProgress: implementingTasks.length,
      pending: pendingTasks.length,
      total: totalTasks,
      percentComplete: Math.round(
        (completedTasks.length / totalTasks) * 100
      ),
    },
    handover: {
      exists: handoverExists,
      currentTaskPath: handoverExists ? currentTaskPath : undefined,
      completionSignalPath: fs.existsSync(completionSignalPath)
        ? completionSignalPath
        : undefined,
    },
    recentActivity,
  };
}

// Register the tool
registerTool(definition, handler);

export { definition, handler };
```

## Test Cases

```typescript
describe('get_context tool', () => {
  it('returns sprint information', async () => {
    const result = await handler({});
    
    expect(result.sprint.id).toBeDefined();
    expect(result.sprint.name).toBeDefined();
    expect(result.sprint.status).toBeDefined();
  });
  
  it('returns null currentTask when no task in progress', async () => {
    // Setup: all tasks PENDING or COMPLETE
    const result = await handler({});
    
    expect(result.currentTask).toBeNull();
  });
  
  it('returns current task when one is in progress', async () => {
    // Setup: one task in IMPLEMENT status
    const result = await handler({});
    
    expect(result.currentTask).not.toBeNull();
    expect(result.currentTask?.status).toBe('IMPLEMENT');
  });
  
  it('calculates progress correctly', async () => {
    // Setup: 3 tasks, 1 complete, 1 in progress, 1 pending
    const result = await handler({});
    
    expect(result.progress.completed).toBe(1);
    expect(result.progress.inProgress).toBe(1);
    expect(result.progress.pending).toBe(1);
    expect(result.progress.total).toBe(3);
    expect(result.progress.percentComplete).toBe(33);
  });
  
  it('returns handover paths when files exist', async () => {
    // Setup: handover files exist
    const result = await handler({});
    
    expect(result.handover.exists).toBe(true);
    expect(result.handover.currentTaskPath).toContain('current-task.md');
  });
});
```
