## Implementation

```typescript
// src/mcp/tools/prepare_task.ts
import { Tool } from '@modelcontextprotocol/sdk/types.js';
import { registerTool } from './registry.js';
import { runPrepare } from '../../core/prepare.js';
import { loadManifest, getTask } from '../../core/manifest.js';
import { OrchestraMcpError, ErrorCode } from '../errors.js';

const definition: Tool = {
  name: 'prepare_task',
  description: 'Prepare the next task handover for implementation. Creates handover documents and updates task status to IMPLEMENT.',
  inputSchema: {
    type: 'object',
    properties: {
      taskId: {
        type: 'number',
        description: 'Specific task ID to prepare. If omitted, prepares next pending task.',
      },
      force: {
        type: 'boolean',
        description: 'Force preparation even if another task is in progress.',
      },
      skipCloseout: {
        type: 'boolean',
        description: 'Skip closeout check for previous task.',
      },
    },
  },
};

interface PrepareTaskParams {
  taskId?: number;
  force?: boolean;
  skipCloseout?: boolean;
}

async function handler(params: PrepareTaskParams) {
  const result = await runPrepare({
    task: params.taskId?.toString(),
    force: params.force ?? false,
    skipCloseout: params.skipCloseout ?? false,
  });
  
  if (!result.success) {
    throw new OrchestraMcpError(
      ErrorCode.ApplicationError,
      result.message,
      result.errors
    );
  }
  
  // Get task details from manifest
  const manifestResult = loadManifest();
  if (!manifestResult.success || !manifestResult.data) {
    throw new OrchestraMcpError(
      ErrorCode.StateError,
      'Failed to load manifest'
    );
  }
  
  const task = getTask(manifestResult.data, result.data.taskId);
  if (!task) {
    throw new OrchestraMcpError(
      ErrorCode.TaskNotFound,
      `Task ${result.data.taskId} not found`
    );
  }
  
  // Resolve dependencies
  const dependencies = (task.dependencies ?? []).map((depId) => {
    const dep = getTask(manifestResult.data!, depId);
    return {
      id: depId,
      title: dep?.title ?? 'Unknown',
      status: dep?.status ?? 'UNKNOWN',
    };
  });
  
  return {
    success: true,
    taskId: task.id,
    title: task.title,
    description: task.description ?? '',
    category: task.category ?? 'INTEGRATION',
    handover: {
      currentTaskPath: result.data.files.find((f: string) => 
        f.includes('current-task.md')
      ) ?? '',
      completionSignalPath: result.data.files.find((f: string) => 
        f.includes('completion-signal.md')
      ) ?? '',
      contextPath: result.data.files.find((f: string) => 
        f.includes('task-context.md')
      ) ?? '',
    },
    dependencies,
    nextSteps: [
      'Read the current-task.md handover document',
      'Implement according to the requirements',
      'Run pre-signal checks before signaling completion',
      'Call signal_complete when done',
    ],
  };
}

// Register the tool
registerTool(definition, handler as any);

export { definition, handler };
```

## Test Cases

```typescript
describe('prepare_task tool', () => {
  it('prepares next pending task when no taskId given', async () => {
    const result = await handler({});
    expect(result.success).toBe(true);
    expect(result.taskId).toBeDefined();
    expect(result.handover.currentTaskPath).toContain('current-task.md');
  });
  
  it('prepares specific task when taskId given', async () => {
    const result = await handler({ taskId: 2 });
    expect(result.taskId).toBe(2);
  });
  
  it('fails when task has unsatisfied dependencies', async () => {
    await expect(handler({ taskId: 3 })).rejects.toMatchObject({
      code: ErrorCode.ApplicationError,
    });
  });
  
  it('succeeds with force when another task in progress', async () => {
    const result = await handler({ taskId: 1, force: true });
    expect(result.success).toBe(true);
  });
});
```
