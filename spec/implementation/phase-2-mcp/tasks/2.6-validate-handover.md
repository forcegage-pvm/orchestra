```typescript
// src/mcp/tools/validate_handover.ts
import { Tool } from '@modelcontextprotocol/sdk/types.js';
import { registerTool } from './registry.js';
import { loadConfig, getResolvedPaths, requireOrchestraRoot } from '../../core/config.js';
import { loadManifest, getTask, getTasksByStatus } from '../../core/manifest.js';
import { OrchestraMcpError, ErrorCode } from '../errors.js';
import * as fs from 'node:fs';
import * as path from 'node:path';

const definition: Tool = {
  name: 'validate_handover',
  description: 'Validate that the handover document is complete before starting work.',
  inputSchema: {
    type: 'object',
    properties: {},
  },
};

interface ValidationCheck {
  check: string;
  passed: boolean;
  message?: string;
}

async function handler(_params: Record<string, unknown>) {
  const orchestraRoot = requireOrchestraRoot();
  const config = loadConfig(orchestraRoot);
  const paths = getResolvedPaths(orchestraRoot, config);
  
  const checks: ValidationCheck[] = [];
  const issues: string[] = [];
  
  // Check 1: Handover folder exists
  const handoverPath = paths.handovers;
  if (!fs.existsSync(handoverPath)) {
    checks.push({
      check: 'Handover folder exists',
      passed: false,
      message: `Folder not found: ${handoverPath}`,
    });
    issues.push('Handover folder does not exist. Run "orchestra prepare" first.');
  } else {
    checks.push({ check: 'Handover folder exists', passed: true });
  }
  
  // Check 2: current-task.md exists
  const currentTaskPath = path.join(handoverPath, 'current-task.md');
  if (!fs.existsSync(currentTaskPath)) {
    checks.push({
      check: 'current-task.md exists',
      passed: false,
      message: 'File not found',
    });
    issues.push('No current-task.md file. Run "orchestra prepare" first.');
    
    return {
      valid: false,
      taskId: null,
      taskTitle: null,
      checks,
      issues,
      handoverPath: null,
    };
  }
  checks.push({ check: 'current-task.md exists', passed: true });
  
  // Read content
  const content = fs.readFileSync(currentTaskPath, 'utf-8');
  
  // Check 3: Parse task ID
  let taskId: number | null = null;
  let taskTitle: string | null = null;
  
  const headerMatch = content.match(/^#\s*Task\s+(\d+):\s*(.+)$/m);
  if (headerMatch) {
    taskId = parseInt(headerMatch[1], 10);
    taskTitle = headerMatch[2].trim();
    checks.push({ check: 'Task ID parseable', passed: true });
  } else {
    checks.push({
      check: 'Task ID parseable',
      passed: false,
      message: 'Could not find "# Task N: Title" header',
    });
    issues.push('Invalid handover format: missing task header');
  }
  
  // Check 4: Has overview section
  if (content.includes('## Overview')) {
    checks.push({ check: 'Has Overview section', passed: true });
  } else {
    checks.push({
      check: 'Has Overview section',
      passed: false,
      message: 'Missing ## Overview section',
    });
    issues.push('Handover missing Overview section');
  }
  
  // Check 5: Has validation reminder
  if (content.includes('BEFORE YOU START') || content.includes('MANDATORY')) {
    checks.push({ check: 'Has validation reminder', passed: true });
  } else {
    checks.push({
      check: 'Has validation reminder',
      passed: false,
      message: 'Missing validation reminder section',
    });
    // Not a blocking issue, just a warning
  }
  
  // Check 6: Task in IMPLEMENT status
  if (taskId !== null) {
    const manifestResult = loadManifest(paths.manifest);
    if (manifestResult.success && manifestResult.data) {
      const task = getTask(manifestResult.data, taskId);
      if (task?.status === 'IMPLEMENT') {
        checks.push({ check: 'Task status is IMPLEMENT', passed: true });
      } else {
        checks.push({
          check: 'Task status is IMPLEMENT',
          passed: false,
          message: `Task status is ${task?.status ?? 'unknown'}`,
        });
        issues.push(`Task ${taskId} is not in IMPLEMENT status`);
      }
    }
  }
  
  // Check 7: Completion signal template exists
  const signalPath = path.join(handoverPath, 'completion-signal.md');
  if (fs.existsSync(signalPath)) {
    checks.push({ check: 'completion-signal.md exists', passed: true });
  } else {
    checks.push({
      check: 'completion-signal.md exists',
      passed: false,
      message: 'Completion signal template missing',
    });
    issues.push('Missing completion-signal.md template');
  }
  
  const valid = checks.every((c) => c.passed);
  
  return {
    valid,
    taskId,
    taskTitle,
    checks,
    issues,
    handoverPath: valid ? currentTaskPath : null,
  };
}

// Register the tool
registerTool(definition, handler);

export { definition, handler };
```

## Test Cases

```typescript
describe('validate_handover tool', () => {
  it('returns valid when handover is complete', async () => {
    // Setup: create valid handover
    const result = await handler({});
    
    expect(result.valid).toBe(true);
    expect(result.taskId).toBeDefined();
    expect(result.issues).toHaveLength(0);
  });
  
  it('returns invalid when no handover exists', async () => {
    // Setup: empty handover folder
    const result = await handler({});
    
    expect(result.valid).toBe(false);
    expect(result.issues).toContain('No current-task.md file');
  });
  
  it('returns invalid when task status wrong', async () => {
    // Setup: handover exists but task is PENDING
    const result = await handler({});
    
    expect(result.valid).toBe(false);
    expect(result.checks.find((c) => c.check.includes('IMPLEMENT'))?.passed).toBe(false);
  });
  
  it('parses task ID and title', async () => {
    const result = await handler({});
    
    expect(result.taskId).toBeGreaterThan(0);
    expect(result.taskTitle).toBeDefined();
  });
});
```
