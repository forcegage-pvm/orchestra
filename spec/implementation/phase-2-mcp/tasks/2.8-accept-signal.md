## Implementation

```typescript
import { runAcceptSignal } from '../../core/index.js';
import { ToolHandler } from '../types.js';

export const acceptSignalTool: ToolHandler = {
  name: 'accept_signal',
  description: 'Validate implementor signal and run pre-verification checks (Orchestrator)',
  inputSchema: {
    type: 'object',
    properties: {
      taskId: {
        type: 'number',
        description: 'Task ID to accept signal for'
      }
    }
  },
  handler: async (params) => {
    const result = await runAcceptSignal({ task: params.taskId?.toString() });
    return {
      accepted: result.accepted,
      taskId: result.taskId,
      signalPath: result.signalPath,
      checks: result.checks,
      nextStep: result.accepted ? 'verify' : 'signal again'
    };
  }
};
```
