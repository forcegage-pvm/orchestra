## Implementation

```typescript
import { runVerification } from '../../core/index.js';
import { ToolHandler } from '../types.js';

export const verifyTool: ToolHandler = {
  name: 'verify',
  description: 'Run hidden verification criteria against task (Orchestrator only)',
  inputSchema: {
    type: 'object',
    properties: {
      taskId: {
        type: 'number',
        description: 'Task ID to verify'
      }
    }
  },
  handler: async (params) => {
    const result = await runVerification({ task: params.taskId?.toString() });
    return {
      success: result.success,
      taskId: result.taskId,
      passed: result.passed,
      results: result.results,  // Hidden - never expose to implementor
      nextStep: result.passed ? 'complete' : 'feedback'
    };
  }
};
```

**WARNING**: This tool returns hidden verification criteria. It should ONLY be used by Orchestrator agents.
